## 决策 1：Faiss ANN — **现在就接入，但分两层**

### 结论：**接入 Faiss，但用最简单的方式**

理由如下：

**（a）不接的风险比接的成本大**

```
你的双条件检索流程：
  Level 1: 在 N_windows 个状态向量中找 top-M=50
  Level 2: 在 M*N_patches_per_window 个上下文键中找 top-K=20

假设训练集有 50000 个窗口，每窗口 16 个 patch（细粒度尺度）：
  Level 1: 50000 次 d_state=128 维距离计算 → 还行
  Level 2: 50*16 = 800 个候选 → 暴力就够了

但如果训练集有 500000 个窗口（SMD 全部 28 台机器、SWaT 长序列）：
  Level 1: 500000 次距离计算 × 每个测试窗口 → 推理直接卡死
```

在论文实验中你迟早要跑大数据集。如果到时候再接 Faiss，检索逻辑要重写，不如现在就把接口留好。

**（b）接入成本其实很低**

Faiss 的 IndexFlatL2 就是精确暴力搜索的 GPU 加速版，**结果和你现在的 torch 版完全一致**，只是快 10-50 倍。所以第一步甚至不需要用近似索引，只需要把存储和检索从 torch tensor 搬到 Faiss index。

### 具体方案：两层策略

```
Level 1（状态粗筛）：
  - 训练集 < 100K 窗口 → IndexFlatL2（精确，够快）
  - 训练集 > 100K 窗口 → IndexIVFFlat（近似，nprobe=16）
  
Level 2（上下文精排）：
  - 候选集 = M × N_patches_per_window ≈ 50×16 = 800
  - 这个规模永远用暴力就行，不需要 Faiss
  - 保持现在的 torch 实现即可
```

所以实际上**只有 Level 1 需要 Faiss，Level 2 保持不变**。

```python
"""
memory/faiss_index.py
Level-1 state retrieval with Faiss; Level-2 stays exact torch.
"""

import numpy as np
import torch
from typing import Tuple

try:
    import faiss
    HAS_FAISS = True
except ImportError:
    HAS_FAISS = False
    print("[Warning] faiss not installed, falling back to exact torch search. "
          "Install with: pip install faiss-cpu  (or faiss-gpu)")


class StateIndex:
    """
    Level-1 state coarse screening.
    Automatically selects:
      - IndexFlatL2 (exact) if n_vectors < 100K
      - IndexIVFFlat (approximate) if n_vectors >= 100K
    Falls back to pure torch if faiss not available.
    """

    def __init__(self, d_state: int, use_gpu: bool = False):
        self.d_state = d_state
        self.use_gpu = use_gpu and HAS_FAISS and faiss.get_num_gpus() > 0
        self.index = None
        self.vectors = None  # keep a copy for torch fallback
        self.n_vectors = 0
        self._use_faiss = HAS_FAISS

    def build(self, state_vectors: np.ndarray):
        """
        Build index from training state vectors.
        Args:
            state_vectors: (N, d_state) float32
        """
        state_vectors = np.ascontiguousarray(state_vectors.astype(np.float32))
        self.n_vectors = len(state_vectors)
        self.vectors = state_vectors

        if not self._use_faiss:
            # Pure torch fallback
            return

        if self.n_vectors < 100_000:
            # Exact search — same results as brute force, just faster
            self.index = faiss.IndexFlatL2(self.d_state)
        else:
            # Approximate search with IVF
            n_clusters = min(int(np.sqrt(self.n_vectors)), 1024)
            quantizer = faiss.IndexFlatL2(self.d_state)
            self.index = faiss.IndexIVFFlat(
                quantizer, self.d_state, n_clusters, faiss.METRIC_L2
            )
            self.index.train(state_vectors)
            self.index.nprobe = 16  # search 16 clusters (balance speed/accuracy)

        self.index.add(state_vectors)

        if self.use_gpu:
            res = faiss.StandardGpuResources()
            self.index = faiss.index_cpu_to_gpu(res, 0, self.index)

        print(f"[StateIndex] Built {'IVF' if self.n_vectors >= 100_000 else 'Flat'} "
              f"index with {self.n_vectors} vectors, d={self.d_state}")

    def search(self, query: np.ndarray, top_m: int) -> Tuple[np.ndarray, np.ndarray]:
        """
        Find top-M nearest state vectors.
        Args:
            query: (B, d_state) float32
        Returns:
            distances: (B, top_m) L2 distances
            indices:   (B, top_m) indices into training set
        """
        query = np.ascontiguousarray(query.astype(np.float32))
        top_m = min(top_m, self.n_vectors)

        if self._use_faiss and self.index is not None:
            distances, indices = self.index.search(query, top_m)
            return distances, indices
        else:
            return self._torch_search(query, top_m)

    def _torch_search(self, query: np.ndarray, top_m: int) -> Tuple[np.ndarray, np.ndarray]:
        """Fallback: exact search with torch."""
        q = torch.from_numpy(query)       # (B, d)
        v = torch.from_numpy(self.vectors) # (N, d)
        # Compute L2 distances
        dists = torch.cdist(q, v, p=2.0)  # (B, N)
        dists_sq = dists ** 2
        topk = torch.topk(dists_sq, top_m, dim=1, largest=False)
        return topk.values.numpy(), topk.indices.numpy()


class ContextSearcher:
    """
    Level-2 context fine ranking.
    Always exact (candidate set is small: M * N_patches ≈ 800).
    Stays in torch for GPU acceleration.
    """

    @staticmethod
    def search(
        query_c: torch.Tensor,
        candidate_c: torch.Tensor,
        candidate_z: torch.Tensor,
        top_k: int,
    ) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        For each query context key, find top-K context-similar patches
        among candidates, then return their z representations.

        Args:
            query_c:     (N_q, d_z)  context keys of current window's patches
            candidate_c: (N_cand, d_z) context keys of candidate patches
            candidate_z: (N_cand, d_z) representations of candidate patches
            top_k:       int

        Returns:
            neighbor_z:     (N_q, top_k, d_z) retrieved patch representations
            neighbor_dists: (N_q, top_k) context distances
        """
        actual_k = min(top_k, candidate_c.shape[0])
        if actual_k == 0:
            d_z = query_c.shape[1]
            return (torch.zeros(query_c.shape[0], top_k, d_z, device=query_c.device),
                    torch.full((query_c.shape[0], top_k), float('inf'), device=query_c.device))

        # L2 distance between queries and candidates
        dists = torch.cdist(query_c, candidate_c, p=2.0)  # (N_q, N_cand)

        # Top-K smallest distances
        topk_dists, topk_idx = torch.topk(dists, actual_k, dim=1, largest=False)

        # Gather neighbor representations
        neighbor_z = candidate_z[topk_idx]  # (N_q, actual_k, d_z)

        # Pad if actual_k < top_k
        if actual_k < top_k:
            pad_n = top_k - actual_k
            d_z = candidate_z.shape[1]
            neighbor_z = torch.cat([
                neighbor_z,
                torch.zeros(query_c.shape[0], pad_n, d_z, device=query_c.device)
            ], dim=1)
            topk_dists = torch.cat([
                topk_dists,
                torch.full((query_c.shape[0], pad_n), float('inf'), device=query_c.device)
            ], dim=1)

        return neighbor_z, topk_dists
```

```python
class MemoryBank:

    def __init__(self, config):
        self.config = config
        # Level-1: Faiss state index
        self.state_index = StateIndex(
            d_state=config.d_state,
            use_gpu=(config.device == 'cuda'),
        )
        # Level-2: exact torch search (unchanged)
        self.context_searcher = ContextSearcher()

        # Storage
        self.patch_z = None       # list of (N_total, d_z) per scale
        self.patch_c = None       # list of (N_total, d_z) per scale
        self.patch_window_map = None  # list of (N_total,) per scale

    def build(
        self,
        state_vectors: np.ndarray,
        patch_z_list: list,
        patch_c_list: list,
        patch_window_maps: list,
        completion_scores: list = None,
    ):
        """Build index. Self-cleaning + Faiss indexing."""
        # ---- Self-cleaning ----
        if completion_scores is not None:
            for l in range(len(patch_z_list)):
                scores = completion_scores[l]
                threshold = np.percentile(scores, 100 * (1 - self.config.clean_ratio))
                keep = scores <= threshold
                patch_z_list[l] = patch_z_list[l][keep]
                patch_c_list[l] = patch_c_list[l][keep]
                patch_window_maps[l] = patch_window_maps[l][keep]

        self.patch_z = patch_z_list
        self.patch_c = patch_c_list
        self.patch_window_map = patch_window_maps

        # ---- Build Level-1 Faiss index ----
        self.state_index.build(state_vectors)
        self.state_vectors = state_vectors

    def retrieve_single_window(
        self,
        g_w: np.ndarray,
        c_list: list,
        z_list: list,
        device: torch.device = torch.device('cpu'),
    ) -> dict:
        """
        Full dual-key retrieval for one test window.

        Args:
            g_w: (d_state,) state vector
            c_list: list of (N_l, d_z) context keys per scale
            z_list: list of (N_l, d_z) patch representations per scale

        Returns:
            dict with 'mem_scores', 'state_novelty', 'coverage' per scale
        """
        # ==== Level 1: Faiss state coarse screening ====
        g_w_2d = g_w.reshape(1, -1)
        distances, indices = self.state_index.search(g_w_2d, self.config.top_M)
        top_m_indices = indices[0]  # (top_M,)

        # State novelty
        k_s = min(self.config.state_novelty_k, len(distances[0]))
        state_novelty = float(np.mean(np.sort(distances[0])[:k_s]))

        # ==== Level 2: Exact context fine ranking (per scale) ====
        mem_scores_list = []
        coverage_list = []

        for l in range(len(c_list)):
            N_l = c_list[l].shape[0]

            # Get candidate patches from coarse-screened windows
            valid_mask = np.isin(self.patch_window_map[l], top_m_indices)
            if valid_mask.sum() == 0:
                mem_scores_list.append(np.full(N_l, 1e6))
                coverage_list.append(np.zeros(N_l))
                continue

            cand_c = torch.from_numpy(self.patch_c[l][valid_mask]).to(device)
            cand_z = torch.from_numpy(self.patch_z[l][valid_mask]).to(device)
            query_c = torch.from_numpy(c_list[l]).to(device)
            query_z = torch.from_numpy(z_list[l]).to(device)

            # Context fine ranking
            neighbor_z, ctx_dists = self.context_searcher.search(
                query_c, cand_c, cand_z, self.config.top_K,
            )

            # kNN distance in representation space
            z_dists = torch.cdist(
                query_z.unsqueeze(1), neighbor_z, p=2.0
            ).squeeze(1)  # (N_l, top_K)

            # Take mean of k nearest
            knn_k = min(self.config.knn_k, z_dists.shape[1])
            topk_z_dists, _ = torch.topk(z_dists, knn_k, dim=1, largest=False)
            mem_score = topk_z_dists.mean(dim=1)  # (N_l,)

            # Coverage: fraction of valid (non-inf) neighbors
            valid_neighbors = (ctx_dists < float('inf')).float()
            coverage = valid_neighbors.mean(dim=1)

            mem_scores_list.append(mem_score.cpu().numpy())
            coverage_list.append(coverage.cpu().numpy())

        return {
            'mem_scores': mem_scores_list,
            'state_novelty': state_novelty,
            'coverage': coverage_list,
        }
```

## 决策 2：Coreset — **保持当前近似版，不换严格版**

### 结论：**不换 k-center-greedy 严格版**

理由：

**（a）严格 k-center-greedy 在你的场景下性价比极低**

```
严格 k-center-greedy 的复杂度：O(n × m × d)
  n = 原始 memory 大小（可能 500K+）
  m = 目标 coreset 大小（比如 50K）
  d = embedding 维度（128）

  = 500K × 50K × 128 ≈ 3.2 × 10^12 次浮点运算

  → 即使在 GPU 上也要跑几十分钟到几小时
  → 而且只在 Stage B 跑一次，不影响推理速度
  → 但如果你要跑 28 个 SMD 子数据集 × 多组超参 × 多次 seed
     这个时间就不可接受了
```

**（b）你的双条件检索本身就有"天然压缩"效果**

关键认识：**你不是在全局 memory 里做 kNN，而是先经过 Level-1 状态筛选（只留 M=50 个窗口），再经过 Level-2 上下文精排（只取 K=20 个 patch）。**

这意味着：即使全局 memory 有 100 万个 patch，实际参与 kNN 距离计算的只有 ~800 个（50 窗口 × ~16 patch）。在这个规模下，coreset 压缩的意义主要是**节省存储**，而不是加速检索。

存储方面，当前的 farthest-point 近似版已经够用了。

**（c）对论文结果几乎没有影响**

严格 k-center-greedy 和 farthest-point 近似的理论保证差异在 2x 以内（近似比）。在实际异常检测的 F1 上，这种差异几乎不可测量——因为你的判别瓶颈不在 coreset 质量上，而在 encoder 表征质量和检索条件设计上。

### 唯一需要做的小改进

当前版本"超大库退化成随机子采样"这个逻辑，建议改成**分层采样**而非纯随机，确保每个状态区域都有代表：

```python
"""
memory/coreset.py — 改进后的大规模 fallback
"""

import numpy as np
from typing import Optional


def farthest_point_sampling(
    features: np.ndarray,
    n_select: int,
    seed: int = 42,
) -> np.ndarray:
    """
    Farthest-point sampling (greedy approximation to k-center).
    Good quality, but O(n * n_select * d).

    Args:
        features: (N, d) float32
        n_select: target coreset size
    Returns:
        indices: (n_select,) selected indices
    """
    N, d = features.shape
    n_select = min(n_select, N)
    if n_select == N:
        return np.arange(N)

    rng = np.random.RandomState(seed)
    selected = [rng.randint(N)]
    min_dists = np.full(N, np.inf)

    for _ in range(1, n_select):
        last = features[selected[-1]]
        dists = np.linalg.norm(features - last, axis=1)
        min_dists = np.minimum(min_dists, dists)
        # Exclude already selected
        min_dists[selected] = -1
        next_idx = np.argmax(min_dists)
        selected.append(next_idx)

    return np.array(selected)


def stratified_random_sampling(
    features: np.ndarray,
    window_ids: np.ndarray,
    n_select: int,
    seed: int = 42,
) -> np.ndarray:
    """
    Stratified random sampling: ensures every source window
    contributes proportionally. Better than pure random for
    preserving state diversity.

    Args:
        features: (N, d)
        window_ids: (N,) which window each patch came from
        n_select: target size
    Returns:
        indices: (n_select,)
    """
    rng = np.random.RandomState(seed)
    N = len(features)
    n_select = min(n_select, N)
    if n_select == N:
        return np.arange(N)

    unique_windows = np.unique(window_ids)
    n_windows = len(unique_windows)

    # Allocate quota per window (proportional to its patch count)
    window_counts = {w: (window_ids == w).sum() for w in unique_windows}
    total = sum(window_counts.values())

    selected = []
    for w in unique_windows:
        w_indices = np.where(window_ids == w)[0]
        # Proportional quota, at least 1
        quota = max(1, int(round(n_select * len(w_indices) / total)))
        quota = min(quota, len(w_indices))
        chosen = rng.choice(w_indices, quota, replace=False)
        selected.extend(chosen.tolist())

    # Adjust to exact n_select
    selected = np.array(selected)
    if len(selected) > n_select:
        selected = rng.choice(selected, n_select, replace=False)
    elif len(selected) < n_select:
        remaining = np.setdiff1d(np.arange(N), selected)
        extra = rng.choice(remaining, n_select - len(selected), replace=False)
        selected = np.concatenate([selected, extra])

    return selected


def build_coreset(
    features: np.ndarray,
    window_ids: np.ndarray,
    max_size: int = 50000,
    fps_threshold: int = 100000,
    seed: int = 42,
) -> np.ndarray:
    """
    Adaptive coreset selection.

    Strategy:
      N <= max_size         → keep all (no compression needed)
      max_size < N <= fps_t → farthest-point sampling (good quality)
      N > fps_threshold     → stratified random sampling (fast, diversity-preserving)

    Args:
        features: (N, d) patch embeddings
        window_ids: (N,) source window index for each patch
        max_size: target coreset size
        fps_threshold: above this size, switch from FPS to stratified random
    Returns:
        indices: (n_selected,)
    """
    N = len(features)

    if N <= max_size:
        print(f"[Coreset] N={N} <= max_size={max_size}, keeping all.")
        return np.arange(N)
    elif N <= fps_threshold:
        print(f"[Coreset] N={N}, using farthest-point sampling -> {max_size}")
        return farthest_point_sampling(features, max_size, seed=seed)
    else:
        print(f"[Coreset] N={N} > {fps_threshold}, using stratified random -> {max_size}")
        return stratified_random_sampling(features, window_ids, max_size, seed=seed)
```

```
  Level 1: 在 N_windows 个状态向量中找 top-M=50
  Level 2: 在 M*N_patches_per_window 个上下文键中找 top-K=20

假设训练集有 50000 个窗口，每窗口 16 个 patch（细粒度尺度）：
  Level 1: 50000 次 d_state=128 维距离计算 → 还行
  Level 2: 50*16 = 800 个候选 → 暴力就够了

但如果训练集有 500000 个窗口（SMD 全部 28 台机器、SWaT 长序列）：
  Level 1: 500000 次距离计算 × 每个测试窗口 → 推理直接卡死
```

在论文实验中你迟早要跑大数据集。如果到时候再接 Faiss，检索逻辑要重写，不如现在就把接口留好。

**（b）接入成本其实很低**

Faiss 的 IndexFlatL2 就是精确暴力搜索的 GPU 加速版，**结果和你现在的 torch 版完全一致**，只是快 10-50 倍。所以第一步甚至不需要用近似索引，只需要把存储和检索从 torch tensor 搬到 Faiss index。

### 具体方案：两层策略

```
Level 1（状态粗筛）：
  - 训练集 < 100K 窗口 → IndexFlatL2（精确，够快）
  - 训练集 > 100K 窗口 → IndexIVFFlat（近似，nprobe=16）
  
Level 2（上下文精排）：
  - 候选集 = M × N_patches_per_window ≈ 50×16 = 800
  - 这个规模永远用暴力就行，不需要 Faiss
  - 保持现在的 torch 实现即可
```

所以实际上**只有 Level 1 需要 Faiss，Level 2 保持不变**。

```python
"""
memory/faiss_index.py
Level-1 state retrieval with Faiss; Level-2 stays exact torch.
"""

import numpy as np
import torch
from typing import Tuple

try:
    import faiss
    HAS_FAISS = True
except ImportError:
    HAS_FAISS = False
    print("[Warning] faiss not installed, falling back to exact torch search. "
          "Install with: pip install faiss-cpu  (or faiss-gpu)")


class StateIndex:
    """
    Level-1 state coarse screening.
    Automatically selects:
      - IndexFlatL2 (exact) if n_vectors < 100K
      - IndexIVFFlat (approximate) if n_vectors >= 100K
    Falls back to pure torch if faiss not available.
    """

    def __init__(self, d_state: int, use_gpu: bool = False):
        self.d_state = d_state
        self.use_gpu = use_gpu and HAS_FAISS and faiss.get_num_gpus() > 0
        self.index = None
        self.vectors = None  # keep a copy for torch fallback
        self.n_vectors = 0
        self._use_faiss = HAS_FAISS

    def build(self, state_vectors: np.ndarray):
        """
        Build index from training state vectors.
        Args:
            state_vectors: (N, d_state) float32
        """
        state_vectors = np.ascontiguousarray(state_vectors.astype(np.float32))
        self.n_vectors = len(state_vectors)
        self.vectors = state_vectors

        if not self._use_faiss:
            # Pure torch fallback
            return

        if self.n_vectors < 100_000:
            # Exact search — same results as brute force, just faster
            self.index = faiss.IndexFlatL2(self.d_state)
        else:
            # Approximate search with IVF
            n_clusters = min(int(np.sqrt(self.n_vectors)), 1024)
            quantizer = faiss.IndexFlatL2(self.d_state)
            self.index = faiss.IndexIVFFlat(
                quantizer, self.d_state, n_clusters, faiss.METRIC_L2
            )
            self.index.train(state_vectors)
            self.index.nprobe = 16  # search 16 clusters (balance speed/accuracy)

        self.index.add(state_vectors)

        if self.use_gpu:
            res = faiss.StandardGpuResources()
            self.index = faiss.index_cpu_to_gpu(res, 0, self.index)

        print(f"[StateIndex] Built {'IVF' if self.n_vectors >= 100_000 else 'Flat'} "
              f"index with {self.n_vectors} vectors, d={self.d_state}")

    def search(self, query: np.ndarray, top_m: int) -> Tuple[np.ndarray, np.ndarray]:
        """
        Find top-M nearest state vectors.
        Args:
            query: (B, d_state) float32
        Returns:
            distances: (B, top_m) L2 distances
            indices:   (B, top_m) indices into training set
        """
        query = np.ascontiguousarray(query.astype(np.float32))
        top_m = min(top_m, self.n_vectors)

        if self._use_faiss and self.index is not None:
            distances, indices = self.index.search(query, top_m)
            return distances, indices
        else:
            return self._torch_search(query, top_m)

    def _torch_search(self, query: np.ndarray, top_m: int) -> Tuple[np.ndarray, np.ndarray]:
        """Fallback: exact search with torch."""
        q = torch.from_numpy(query)       # (B, d)
        v = torch.from_numpy(self.vectors) # (N, d)
        # Compute L2 distances
        dists = torch.cdist(q, v, p=2.0)  # (B, N)
        dists_sq = dists ** 2
        topk = torch.topk(dists_sq, top_m, dim=1, largest=False)
        return topk.values.numpy(), topk.indices.numpy()


class ContextSearcher:
    """
    Level-2 context fine ranking.
    Always exact (candidate set is small: M * N_patches ≈ 800).
    Stays in torch for GPU acceleration.
    """

    @staticmethod
    def search(
        query_c: torch.Tensor,
        candidate_c: torch.Tensor,
        candidate_z: torch.Tensor,
        top_k: int,
    ) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        For each query context key, find top-K context-similar patches
        among candidates, then return their z representations.

        Args:
            query_c:     (N_q, d_z)  context keys of current window's patches
            candidate_c: (N_cand, d_z) context keys of candidate patches
            candidate_z: (N_cand, d_z) representations of candidate patches
            top_k:        int

        Returns:
            neighbor_z:     (N_q, top_k, d_z) retrieved patch representations
            neighbor_dists: (N_q, top_k) context distances
        """
        actual_k = min(top_k, candidate_c.shape[0])
        if actual_k == 0:
            d_z = query_c.shape[1]
            return (torch.zeros(query_c.shape[0], top_k, d_z, device=query_c.device),
                    torch.full((query_c.shape[0], top_k), float('inf'), device=query_c.device))

        # L2 distance between queries and candidates
        dists = torch.cdist(query_c, candidate_c, p=2.0)  # (N_q, N_cand)

        # Top-K smallest distances
        topk_dists, topk_idx = torch.topk(dists, actual_k, dim=1, largest=False)

        # Gather neighbor representations
        neighbor_z = candidate_z[topk_idx]  # (N_q, actual_k, d_z)

        # Pad if actual_k < top_k
        if actual_k < top_k:
            pad_n = top_k - actual_k
            d_z = candidate_z.shape[1]
            neighbor_z = torch.cat([
                neighbor_z,
                torch.zeros(query_c.shape[0], pad_n, d_z, device=query_c.device)
            ], dim=1)
            topk_dists = torch.cat([
                topk_dists,
                torch.full((query_c.shape[0], pad_n), float('inf'), device=query_c.device)
            ], dim=1)

        return neighbor_z, topk_dists
```

```python
"""
memory/memory_bank.py — 修改后的 build() 和 retrieve()
只需改 Level-1 部分，Level-2 不变。
"""

from memory.faiss_index import StateIndex, ContextSearcher


class MemoryBank:

    def __init__(self, config):
        self.config = config
        # Level-1: Faiss state index
        self.state_index = StateIndex(
            d_state=config.d_state,
            use_gpu=(config.device == 'cuda'),
        )
        # Level-2: exact torch search (unchanged)
        self.context_searcher = ContextSearcher()

        # Storage
        self.patch_z = None       # list of (N_total, d_z) per scale
        self.patch_c = None       # list of (N_total, d_z) per scale
        self.patch_window_map = None  # list of (N_total,) per scale

    def build(
        self,
        state_vectors: np.ndarray,
        patch_z_list: list,
        patch_c_list: list,
        patch_window_maps: list,
        completion_scores: list = None,
    ):
        """Build index. Self-cleaning + Faiss indexing."""
        # ---- Self-cleaning ----
        if completion_scores is not None:
            for l in range(len(patch_z_list)):
                scores = completion_scores[l]
                threshold = np.percentile(scores, 100 * (1 - self.config.clean_ratio))
                keep = scores <= threshold
                patch_z_list[l] = patch_z_list[l][keep]
                patch_c_list[l] = patch_c_list[l][keep]
                patch_window_maps[l] = patch_window_maps[l][keep]

        self.patch_z = patch_z_list
        self.patch_c = patch_c_list
        self.patch_window_map = patch_window_maps

        # ---- Build Level-1 Faiss index ----
        self.state_index.build(state_vectors)
        self.state_vectors = state_vectors

    def retrieve_single_window(
        self,
        g_w: np.ndarray,
        c_list: list,
        z_list: list,
        device: torch.device = torch.device('cpu'),
    ) -> dict:
        """
        Full dual-key retrieval for one test window.

        Args:
            g_w: (d_state,) state vector
            c_list: list of (N_l, d_z) context keys per scale
            z_list: list of (N_l, d_z) patch representations per scale

        Returns:
            dict with 'mem_scores', 'state_novelty', 'coverage' per scale
        """
        # ==== Level 1: Faiss state coarse screening ====
        g_w_2d = g_w.reshape(1, -1)
        distances, indices = self.state_index.search(g_w_2d, self.config.top_M)
        top_m_indices = indices[0]  # (top_M,)

        # State novelty
        k_s = min(self.config.state_novelty_k, len(distances[0]))
        state_novelty = float(np.mean(np.sort(distances[0])[:k_s]))

        # ==== Level 2: Exact context fine ranking (per scale) ====
        mem_scores_list = []
        coverage_list = []

        for l in range(len(c_list)):
            N_l = c_list[l].shape[0]

            # Get candidate patches from coarse-screened windows
            valid_mask = np.isin(self.patch_window_map[l], top_m_indices)
            if valid_mask.sum() == 0:
                mem_scores_list.append(np.full(N_l, 1e6))
                coverage_list.append(np.zeros(N_l))
                continue

            cand_c = torch.from_numpy(self.patch_c[l][valid_mask]).to(device)
            cand_z = torch.from_numpy(self.patch_z[l][valid_mask]).to(device)
            query_c = torch.from_numpy(c_list[l]).to(device)
            query_z = torch.from_numpy(z_list[l]).to(device)

            # Context fine ranking
            neighbor_z, ctx_dists = self.context_searcher.search(
                query_c, cand_c, cand_z, self.config.top_K,
            )

            # kNN distance in representation space
            z_dists = torch.cdist(
                query_z.unsqueeze(1), neighbor_z, p=2.0
            ).squeeze(1)  # (N_l, top_K)

            # Take mean of k nearest
            knn_k = min(self.config.knn_k, z_dists.shape[1])
            topk_z_dists, _ = torch.topk(z_dists, knn_k, dim=1, largest=False)
            mem_score = topk_z_dists.mean(dim=1)  # (N_l,)

            # Coverage: fraction of valid (non-inf) neighbors
            valid_neighbors = (ctx_dists < float('inf')).float()
            coverage = valid_neighbors.mean(dim=1)

            mem_scores_list.append(mem_score.cpu().numpy())
            coverage_list.append(coverage.cpu().numpy())

        return {
            'mem_scores': mem_scores_list,
            'state_novelty': state_novelty,
            'coverage': coverage_list,
        }
```

## 决策 2：Coreset — **保持当前近似版，不换严格版**

### 结论：**不换 k-center-greedy 严格版**

理由：

**（a）严格 k-center-greedy 在你的场景下性价比极低**

```
严格 k-center-greedy 的复杂度：O(n × m × d)
  n = 原始 memory 大小（可能 500K+）
  m = 目标 coreset 大小（比如 50K）
  d = embedding 维度（128）

  = 500K × 50K × 128 ≈ 3.2 × 10^12 次浮点运算

  → 即使在 GPU 上也要跑几十分钟到几小时
  → 而且只在 Stage B 跑一次，不影响推理速度
  → 但如果你要跑 28 个 SMD 子数据集 × 多组超参 × 多次 seed
      这个时间就不可接受了
```

**（b）你的双条件检索本身就有"天然压缩"效果**

关键认识：**你不是在全局 memory 里做 kNN，而是先经过 Level-1 状态筛选（只留 M=50 个窗口），再经过 Level-2 上下文精排（只取 K=20 个 patch）。**

这意味着：即使全局 memory 有 100 万个 patch，实际参与 kNN 距离计算的只有 ~800 个（50 窗口 × ~16 patch）。在这个规模下，coreset 压缩的意义主要是**节省存储**，而不是加速检索。

存储方面，当前的 farthest-point 近似版已经够用了。

**（c）对论文结果几乎没有影响**

严格 k-center-greedy 和 farthest-point 近似的理论保证差异在 2x 以内（近似比）。在实际异常检测的 F1 上，这种差异几乎不可测量——因为你的判别瓶颈不在 coreset 质量上，而在 encoder 表征质量和检索条件设计上。

### 唯一需要做的小改进

当前版本"超大库退化成随机子采样"这个逻辑，建议改成**分层采样**而非纯随机，确保每个状态区域都有代表：

```python
"""
memory/coreset.py — 改进后的大规模 fallback
"""

import numpy as np
from typing import Optional


def farthest_point_sampling(
    features: np.ndarray,
    n_select: int,
    seed: int = 42,
) -> np.ndarray:
    """
    Farthest-point sampling (greedy approximation to k-center).
    Good quality, but O(n * n_select * d).

    Args:
        features: (N, d) float32
        n_select: target coreset size
    Returns:
        indices: (n_select,) selected indices
    """
    N, d = features.shape
    n_select = min(n_select, N)
    if n_select == N:
        return np.arange(N)

    rng = np.random.RandomState(seed)
    selected = [rng.randint(N)]
    min_dists = np.full(N, np.inf)

    for _ in range(1, n_select):
        last = features[selected[-1]]
        dists = np.linalg.norm(features - last, axis=1)
        min_dists = np.minimum(min_dists, dists)
        # Exclude already selected
        min_dists[selected] = -1
        next_idx = np.argmax(min_dists)
        selected.append(next_idx)

    return np.array(selected)


def stratified_random_sampling(
    features: np.ndarray,
    window_ids: np.ndarray,
    n_select: int,
    seed: int = 42,
) -> np.ndarray:
    """
    Stratified random sampling: ensures every source window
    contributes proportionally. Better than pure random for
    preserving state diversity.

    Args:
        features: (N, d)
        window_ids: (N,) which window each patch came from
        n_select: target size
    Returns:
        indices: (n_select,)
    """
    rng = np.random.RandomState(seed)
    N = len(features)
    n_select = min(n_select, N)
    if n_select == N:
        return np.arange(N)

    unique_windows = np.unique(window_ids)
    n_windows = len(unique_windows)

    # Allocate quota per window (proportional to its patch count)
    window_counts = {w: (window_ids == w).sum() for w in unique_windows}
    total = sum(window_counts.values())

    selected = []
    for w in unique_windows:
        w_indices = np.where(window_ids == w)[0]
        # Proportional quota, at least 1
        quota = max(1, int(round(n_select * len(w_indices) / total)))
        quota = min(quota, len(w_indices))
        chosen = rng.choice(w_indices, quota, replace=False)
        selected.extend(chosen.tolist())

    # Adjust to exact n_select
    selected = np.array(selected)
    if len(selected) > n_select:
        selected = rng.choice(selected, n_select, replace=False)
    elif len(selected) < n_select:
        remaining = np.setdiff1d(np.arange(N), selected)
        extra = rng.choice(remaining, n_select - len(selected), replace=False)
        selected = np.concatenate([selected, extra])

    return selected


def build_coreset(
    features: np.ndarray,
    window_ids: np.ndarray,
    max_size: int = 50000,
    fps_threshold: int = 100000,
    seed: int = 42,
) -> np.ndarray:
    """
    Adaptive coreset selection.

    Strategy:
      N <= max_size         → keep all (no compression needed)
      max_size < N <= fps_t → farthest-point sampling (good quality)
      N > fps_threshold     → stratified random sampling (fast, diversity-preserving)

    Args:
        features: (N, d) patch embeddings
        window_ids: (N,) source window index for each patch
        max_size: target coreset size
        fps_threshold: above this size, switch from FPS to stratified random
    Returns:
        indices: (n_selected,)
    """
    N = len(features)

    if N <= max_size:
        print(f"[Coreset] N={N} <= max_size={max_size}, keeping all.")
        return np.arange(N)
    elif N <= fps_threshold:
        print(f"[Coreset] N={N}, using farthest-point sampling -> {max_size}")
        return farthest_point_sampling(features, max_size, seed=seed)
    else:
        print(f"[Coreset] N={N} > {fps_threshold}, using stratified random -> {max_size}")
        return stratified_random_sampling(features, window_ids, max_size, seed=seed)
```

## 总结

| 决策 | 结论 | 理由 |
|------|------|------|
| **Faiss** | ✅ **现在接入** | 只改 Level-1（5 行核心代码）；Level-2 不变；避免大数据集推理卡死 |
| **Coreset** | ❌ **不换严格版** | 双条件检索已天然压缩候选集；FPS 近似版 + 分层随机 fallback 足够；严格版耗时太长且对 F1 无可测量影响 |

**一句话：Faiss 是推理性能瓶颈，不接会死；Coreset 精度差异在你的框架里被双条件检索吸收了，不值得为它多花 10 倍的 Stage-B 时间。**
