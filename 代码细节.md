class SlidingWindowDataset(Dataset):
    """
    Sliding window dataset for time series.
    Produces windows of shape (L, C).
    For training: only normal data, labels are all zeros.
    For testing: may contain anomalies, labels are point-level.
    """

    def __init__(
        self,
        data: np.ndarray,
        labels: Optional[np.ndarray],
        seq_len: int,
        stride: int = 1,
    ):
        """
        Args:
            data: (T, C) normalized time series
            labels: (T,) binary labels, 0=normal, 1=anomaly. None for training.
            seq_len: window length
            stride: sliding stride
        """
        self.data = data.astype(np.float32)
        self.labels = labels
        self.seq_len = seq_len
        self.stride = stride

        self.n_windows = (len(data) - seq_len) // stride + 1

    def __len__(self):
        return self.n_windows

    def __getitem__(self, idx):
        start = idx * self.stride
        end = start + self.seq_len
        window = self.data[start:end]  # (L, C)

        result = {
            'x': torch.from_numpy(window),
            'idx': idx,
            'start': start,
        }

        if self.labels is not None:
            window_labels = self.labels[start:end]
            result['labels'] = torch.from_numpy(window_labels.astype(np.float32))

        return result

class STSDDecomposer(nn.Module):
    """
    Slow-Trend / Fast-Residual decomposition.

    Pipeline:
        1. FFT -> learnable soft low-pass mask -> IFFT -> S_0  (frequency-domain init)
        2. Lightweight 1D-Conv correction -> S = S_0 + Delta    (time-domain refine)
        3. R = X - S                                            (residual)

    Not a hard prerequisite: downstream modules have bypass from raw X
    to remain robust against imperfect decomposition.

    Regularization: only L_smooth = |nabla_t^2 S|_1
    """

    def __init__(self, n_channels: int, seq_len: int, d_hidden: int = 64,
                 lowpass_center: float = 0.25):
        super().__init__()
        self.seq_len = seq_len
        self.n_channels = n_channels

        # --- Learnable soft low-pass mask in frequency domain ---
        n_freq = seq_len // 2 + 1
        self.freq_mask_logits = nn.Parameter(torch.zeros(1, n_channels, n_freq))

        # Initialize: sigmoid centered at lowpass_center fraction of Nyquist
        with torch.no_grad():
            freqs = torch.linspace(0, 1, n_freq)
            init_val = 4.0 * (lowpass_center - freqs)  # smooth roll-off
            self.freq_mask_logits.copy_(
                init_val.unsqueeze(0).unsqueeze(0).expand(1, n_channels, -1)
            )

        # --- Time-domain correction network ---
        self.correction = nn.Sequential(
            nn.Conv1d(n_channels * 2, d_hidden, kernel_size=7, padding=3),
            nn.GELU(),
            nn.Conv1d(d_hidden, d_hidden, kernel_size=5, padding=2),
            nn.GELU(),
            nn.Conv1d(d_hidden, n_channels, kernel_size=3, padding=1),
        )

    def forward(self, x: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        Args:
            x: (B, L, C)
        Returns:
            S: (B, L, C) slow component
            R: (B, L, C) fast residual
        """
        B, L, C = x.shape

        # Step 1: Frequency-domain low-pass
        x_freq = torch.fft.rfft(x, dim=1)           # (B, n_freq, C)
        x_freq = x_freq.permute(0, 2, 1)             # (B, C, n_freq)
        mask = torch.sigmoid(self.freq_mask_logits)   # (1, C, n_freq)
        s0_freq = x_freq * mask
        s0_freq = s0_freq.permute(0, 2, 1)            # (B, n_freq, C)
        S0 = torch.fft.irfft(s0_freq, n=L, dim=1)    # (B, L, C)

        # Step 2: Time-domain correction
        corr_input = torch.cat([x, S0], dim=-1).permute(0, 2, 1)  # (B, 2C, L)
        delta = self.correction(corr_input).permute(0, 2, 1)       # (B, L, C)
        S = S0 + delta

        # Step 3: Residual
        R = x - S

        return S, R

    def smoothness_loss(self, S: torch.Tensor) -> torch.Tensor:
        """L_smooth = |nabla_t^2 S|_1"""
        d2S = S[:, 2:, :] - 2 * S[:, 1:-1, :] + S[:, :-2, :]
        return d2S.abs().mean()

        g_w = self.mlp(h)                    # (B, d_state)
        return g_w

        )

        """

        return r_patches * a_w[:, None, None, :]

class STSDDecomposer(nn.Module):
    """
    Slow-Trend / Fast-Residual decomposition.

    Pipeline:
        1. FFT -> learnable soft low-pass mask -> IFFT -> S_0  (frequency-domain init)
        2. Lightweight 1D-Conv correction -> S = S_0 + Delta    (time-domain refine)
        3. R = X - S                                            (residual)

    Not a hard prerequisite: downstream modules have bypass from raw X
    to remain robust against imperfect decomposition.

    Regularization: only L_smooth = |nabla_t^2 S|_1
    """

    def __init__(self, n_channels: int, seq_len: int, d_hidden: int = 64,
                 lowpass_center: float = 0.25):
        super().__init__()
        self.seq_len = seq_len
        self.n_channels = n_channels

        # --- Learnable soft low-pass mask in frequency domain ---
        n_freq = seq_len // 2 + 1
        self.freq_mask_logits = nn.Parameter(torch.zeros(1, n_channels, n_freq))

        # Initialize: sigmoid centered at lowpass_center fraction of Nyquist
        with torch.no_grad():
            freqs = torch.linspace(0, 1, n_freq)
            init_val = 4.0 * (lowpass_center - freqs)  # smooth roll-off
            self.freq_mask_logits.copy_(
                init_val.unsqueeze(0).unsqueeze(0).expand(1, n_channels, -1)
            )

        # --- Time-domain correction network ---
        self.correction = nn.Sequential(
            nn.Conv1d(n_channels * 2, d_hidden, kernel_size=7, padding=3),
            nn.GELU(),
            nn.Conv1d(d_hidden, d_hidden, kernel_size=5, padding=2),
            nn.GELU(),
            nn.Conv1d(d_hidden, n_channels, kernel_size=3, padding=1),
        )

    def forward(self, x: torch.Tensor) -> Tuple[torch.Tensor, torch.Tensor]:
        """
        Args:
            x: (B, L, C)
        Returns:
            S: (B, L, C) slow component
            R: (B, L, C) fast residual
        """
        B, L, C = x.shape

        # Step 1: Frequency-domain low-pass
        x_freq = torch.fft.rfft(x, dim=1)           # (B, n_freq, C)
        x_freq = x_freq.permute(0, 2, 1)             # (B, C, n_freq)
        mask = torch.sigmoid(self.freq_mask_logits)   # (1, C, n_freq)
        s0_freq = x_freq * mask
        s0_freq = s0_freq.permute(0, 2, 1)            # (B, n_freq, C)
        S0 = torch.fft.irfft(s0_freq, n=L, dim=1)    # (B, L, C)

        # Step 2: Time-domain correction
        corr_input = torch.cat([x, S0], dim=-1).permute(0, 2, 1)  # (B, 2C, L)
        delta = self.correction(corr_input).permute(0, 2, 1)       # (B, L, C)
        S = S0 + delta

        # Step 3: Residual
        R = x - S

        return S, R

    def smoothness_loss(self, S: torch.Tensor) -> torch.Tensor:
        """L_smooth = |nabla_t^2 S|_1"""
        d2S = S[:, 2:, :] - 2 * S[:, 1:-1, :] + S[:, :-2, :]
        return d2S.abs().mean()

        super().__init__()
        self.config = config
        self.patch_sizes = config.patch_sizes
        self.n_scales = len(config.patch_sizes)

        # Channel modulation
        self.channel_mod = ChannelModulation(config.d_state, config.n_channels)

        # Shared Conv trunk
        self.shared_trunk = SharedConvTrunk(config.n_channels, config.d_trunk)

        # Scale-specific heads
        self.scale_heads = nn.ModuleList([
            ScaleHead(config.d_trunk, config.d_z) for _ in self.patch_sizes
        ])

        # Bypass encoders
        self.bypass_encoders = nn.ModuleList([
            BypassEncoder(config.n_channels, config.d_bypass) for _ in self.patch_sizes
        ])

        # Bypass -> d_z projection
        self.bypass_projs = nn.ModuleList([
            nn.Linear(config.d_bypass, config.d_z) for _ in self.patch_sizes
        ])

        # Sigmoid-bounded bypass gates (learnable per scale)
        self.bypass_gate_logits = nn.ParameterList([
            nn.Parameter(torch.full((config.d_z,), -2.0))  # init small -> sigmoid ~ 0.12
            for _ in self.patch_sizes
        ])

    @staticmethod
    def make_patches(seq: torch.Tensor, patch_size: int) -> torch.Tensor:
        """
        Non-overlapping patch extraction.
        Args:    seq: (B, L, C)
        Returns: (B, N, P, C) where N = L // P
        """
        B, L, C = seq.shape
        N = L // patch_size
        seq_trimmed = seq[:, :N * patch_size, :]
        return seq_trimmed.reshape(B, N, patch_size, C)

    def forward(
        self, R: torch.Tensor, X: torch.Tensor, g_w: torch.Tensor
    ) -> Dict[str, List[torch.Tensor]]:
        """
        Args:
            R: (B, L, C) residual
            X: (B, L, C) raw input
            g_w: (B, d_state) state vector
        Returns:
            dict with:
              'z':          List[(B, N_l, d_z)]   patch representations
              'u':          List[(B, N_l, d_z)]   fused embeddings
              'c':          List[(B, N_l, d_z)]   context keys
              'r_patches': List[(B, N_l, P_l, C)] residual patches (modulated)
              'n_patches': List[int]              patch count per scale
        """
        B = R.shape[0]
        out = {k: [] for k in ['z', 'u', 'c', 'r_patches', 'n_patches']}

        for l, ps in enumerate(self.patch_sizes):
            # --- Extract patches ---
            r_p = self.make_patches(R, ps)   # (B, N, P, C)
            x_p = self.make_patches(X, ps)   # (B, N, P, C)
            N = r_p.shape[1]

            # --- Channel modulation ---
            r_mod = self.channel_mod(r_p, g_w)  # (B, N, P, C)

            # --- Residual main path ---
            r_flat = r_mod.reshape(B * N, ps, self.config.n_channels)
            trunk_feat = self.shared_trunk(r_flat)       # (B*N, d_trunk)
            e_r = self.scale_heads[l](trunk_feat)        # (B*N, d_z)

            # --- Bypass path ---
            x_flat = x_p.reshape(B * N, ps, self.config.n_channels)
            e_x = self.bypass_encoders[l](x_flat)        # (B*N, d_bypass)
            e_x_proj = self.bypass_projs[l](e_x)         # (B*N, d_z)

            # --- Sigmoid-bounded fusion ---
            alpha = torch.sigmoid(self.bypass_gate_logits[l])  # (d_z,)
            u = e_r + alpha.unsqueeze(0) * e_x_proj            # (B*N, d_z)

            # Reshape
            u = u.reshape(B, N, self.config.d_z)
            z = u  # z = u in this version

            # --- Context key: neighborhood average pooling ---
            c = self._build_context_keys(u)   # (B, N, d_z)

            out['z'].append(z)
            out['u'].append(u)
            out['c'].append(c)
            out['r_patches'].append(r_mod)
            out['n_patches'].append(N)

        return out

    def _build_context_keys(self, u: torch.Tensor) -> torch.Tensor:
        """
        Neighborhood average pooling for context keys.
        Args:    u: (B, N, d_z)
        Returns: c: (B, N, d_z)
        """
        k = self.config.context_k
        B, N, D = u.shape
        # Pad boundaries with zeros
        u_pad = F.pad(u, (0, 0, k, k), mode='constant', value=0.0)
        # Create count mask for proper averaging at boundaries
        ones = torch.ones(B, N, 1, device=u.device)
        ones_pad = F.pad(ones, (0, 0, k, k), mode='constant', value=0.0)

        # Unfold and average
        u_unfold = u_pad.unfold(1, 2 * k + 1, 1)       # (B, N, D, 2k+1)
        ones_unfold = ones_pad.unfold(1, 2 * k + 1, 1)  # (B, N, 1, 2k+1)

        c = u_unfold.sum(dim=-1) / ones_unfold.sum(dim=-1).clamp(min=1)
        return c

class PositionalEncoding(nn.Module):
    """Sinusoidal positional encoding for patch sequences."""

    def __init__(self, d_model: int, max_len: int = 1024):
        super().__init__()
        pe = torch.zeros(max_len, d_model)
        pos = torch.arange(0, max_len).unsqueeze(1).float()
        div = torch.exp(torch.arange(0, d_model, 2).float() * (-math.log(10000.0) / d_model))
        pe[:, 0::2] = torch.sin(pos * div)
        pe[:, 1::2] = torch.cos(pos * div)
        self.register_buffer('pe', pe.unsqueeze(0))  # (1, max_len, d_model)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        return x + self.pe[:, :x.size(1), :]


class MaskedCompletionHead(nn.Module):
    """
    2-layer Transformer for masked patch completion.
    Receives a patch sequence with some positions replaced by [MASK],
    predicts the original residual at those positions.

    Training: random mask with ratio rho
    Testing:  deterministic group mask (index % K), each patch predicted exactly once
    """

    def __init__(self, config: CoReMADConfig, patch_size: int):
        super().__init__()
        self.d_z = config.d_z
        self.patch_size = patch_size
        self.n_channels = config.n_channels
        self.output_dim = patch_size * config.n_channels

        # Learnable mask token
        self.mask_token = nn.Parameter(torch.randn(1, 1, config.d_z) * 0.02)

        # Positional encoding
        self.pos_enc = PositionalEncoding(config.d_z)

        # Transformer encoder
        layer = nn.TransformerEncoderLayer(
            d_model=config.d_z,
            nhead=config.completion_n_heads,
            dim_feedforward=config.d_z * 2,
            dropout=config.completion_dropout,
            batch_first=True,
            activation='gelu',
        )
        self.transformer = nn.TransformerEncoder(layer, num_layers=config.completion_n_layers)

        # Output projection
        self.out_proj = nn.Sequential(
            nn.Linear(config.d_z, config.d_z),
            nn.GELU(),
            nn.Linear(config.d_z, self.output_dim),
        )

    def forward(
        self, u: torch.Tensor, mask: torch.Tensor
    ) -> torch.Tensor:
        """
        Args:
            u: (B, N, d_z) patch embeddings
            mask: (B, N) bool tensor, True = masked
        Returns:
            r_hat: (B, N, P*C) predictions for ALL positions (only masked ones used in loss)
        """
        B, N, D = u.shape

        # Replace masked positions with mask token
        mask_expanded = mask.unsqueeze(-1).expand_as(u)  # (B, N, D)
        u_masked = u.clone()
        u_masked[mask_expanded] = self.mask_token.expand(B, N, -1)[mask_expanded]

        # Add positional encoding
        u_masked = self.pos_enc(u_masked)

        # Transformer
        h = self.transformer(u_masked)  # (B, N, d_z)

        # Project all positions
        r_hat = self.out_proj(h)  # (B, N, P*C)
        return r_hat


class NextPatchPredHead(nn.Module):
    """Predict next patch from current. Training only, not used in scoring."""

    def __init__(self, config: CoReMADConfig, patch_size: int):
        super().__init__()
        self.output_dim = patch_size * config.n_channels
        self.pred = nn.Sequential(
            nn.Linear(config.d_z, config.d_z),
            nn.GELU(),
            nn.Linear(config.d_z, self.output_dim),
        )

    def forward(self, u: torch.Tensor) -> torch.Tensor:
        """u: (B, N, d_z) -> r_hat: (B, N-1, P*C)"""
        return self.pred(u[:, :-1, :])
