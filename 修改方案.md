### 决策 2：纯推理模式必须支持，不依赖训练原始数据

**结论：必须做到。**

理由：

```
论文复现角度：
  - reviewer 或读者拿到你的 checkpoint 后，应该能直接跑 test
  - 如果还要带着训练集一起才能 normalize，等于 checkpoint 不是自包含的

工业部署角度：
  - 部署时不可能把训练集搬到线上
  - normalizer 的 mean/std 必须固化在 checkpoint 里

你当前的 bug：
  - load_stage_a_model() 读回了 normalizer 但后续没用
  - 建库 / 测试 / 阶段C 都重新 prepare_data() 重新 fit
  - 这意味着如果训练集不在，后续阶段全部崩溃
```

**修法：**


### 决策 3：Faiss — 先装 faiss-cpu，GPU 版看服务器情况

**结论：`faiss-cpu` 必装，`faiss-gpu` 可选。**

```
原因：
  - Level-1 检索的规模（最大几十万向量）用 faiss-cpu 就足够快
  - faiss-gpu 的安装依赖 CUDA 版本，经常和 PyTorch 的 CUDA 冲突
  - 你的计算瓶颈不在检索上（4次 mask completion 的前向传播才是瓶颈）
  
推荐：
  pip install faiss-cpu
  
  如果服务器 CUDA 版本和 faiss-gpu 兼容（CUDA 11.x 或 12.x）：
  pip install faiss-gpu
  否则不要强装，faiss-cpu 够用
```

**确保你的代码在 faiss 不可用时有 torch fallback（你现在已经有了），这样本地调试也不受影响。**

```python
# ==========================================
# memory.py — 修改 Level-2 检索和 kNN 距离计算
# ==========================================

def _compute_memory_score(
    self,
    query_z: torch.Tensor,       # (N_l, d_z)
    neighbor_z: torch.Tensor,    # (N_l, top_K, d_z)
    neighbor_valid: torch.Tensor, # (N_l, top_K) bool — 新增！
    knn_k: int = 5,
) -> torch.Tensor:
    """
    Compute kNN distance in z-space, IGNORING padded neighbors.
    
    Args:
        query_z: (N_l, d_z) current patches
        neighbor_z: (N_l, top_K, d_z) retrieved neighbors (may contain zero-padding)
        neighbor_valid: (N_l, top_K) True = real neighbor, False = padding
        knn_k: number of nearest neighbors to average
    
    Returns:
        mem_scores: (N_l,) average kNN distance
    """
    # Compute z-space distances
    # query_z: (N_l, 1, d_z), neighbor_z: (N_l, top_K, d_z)
    z_dists = torch.cdist(
        query_z.unsqueeze(1), neighbor_z, p=2.0
    ).squeeze(1)  # (N_l, top_K)
    
    # Mask out padded neighbors with infinity
    z_dists = z_dists.masked_fill(~neighbor_valid, float('inf'))
    
    # Count actual valid neighbors per query
    n_valid = neighbor_valid.sum(dim=1)  # (N_l,)
    
    # Adaptive k: use min(knn_k, n_valid) for each query
    # Sort distances (inf will be at the end)
    z_dists_sorted, _ = torch.sort(z_dists, dim=1)  # (N_l, top_K)
    
    # Take top-k, but handle cases where n_valid < knn_k
    actual_k = torch.clamp(n_valid, min=1, max=knn_k)  # (N_l,)
    
    # Compute mean of top-k distances for each query
    mem_scores = torch.zeros(query_z.shape[0], device=query_z.device)
    for k_val in range(1, knn_k + 1):
        # For queries with actual_k >= k_val, accumulate the k_val-th distance
        mask = actual_k >= k_val
        if mask.any():
            mem_scores[mask] += z_dists_sorted[mask, k_val - 1]
    
    mem_scores = mem_scores / actual_k.float().clamp(min=1)
    
    # Fallback: if no valid neighbors at all, assign high score
    no_neighbor = n_valid == 0
    if no_neighbor.any():
        mem_scores[no_neighbor] = z_dists_sorted[:, 0].max() * 2  # 高于最大距离
    
    return mem_scores

        """
        Returns:
            neighbor_z:     (N_q, top_k, d_z)
            neighbor_dists: (N_q, top_k)

        """
        actual_k = min(top_k, candidate_c.shape[0])
        N_q = query_c.shape[0]
```

### 必修 3：Level-2 padded neighbor 污染 kNN

```python
# ==========================================
# memory.py — 修改 Level-2 检索和 kNN 距离计算
# ==========================================

def _compute_memory_score(
    self,
    query_z: torch.Tensor,       # (N_l, d_z)
    neighbor_z: torch.Tensor,    # (N_l, top_K, d_z)
    neighbor_valid: torch.Tensor, # (N_l, top_K) bool — 新增！
    knn_k: int = 5,
) -> torch.Tensor:
    """
    Compute kNN distance in z-space, IGNORING padded neighbors.
    
    Args:
        query_z: (N_l, d_z) current patches
        neighbor_z: (N_l, top_K, d_z) retrieved neighbors (may contain zero-padding)
        neighbor_valid: (N_l, top_K) True = real neighbor, False = padding
        knn_k: number of nearest neighbors to average
    
    Returns:
        mem_scores: (N_l,) average kNN distance
    """
    # Compute z-space distances
    # query_z: (N_l, 1, d_z), neighbor_z: (N_l, top_K, d_z)
    z_dists = torch.cdist(
        query_z.unsqueeze(1), neighbor_z, p=2.0
    ).squeeze(1)  # (N_l, top_K)
    
    # Mask out padded neighbors with infinity
    z_dists = z_dists.masked_fill(~neighbor_valid, float('inf'))
    
    # Count actual valid neighbors per query
    n_valid = neighbor_valid.sum(dim=1)  # (N_l,)
    
    # Adaptive k: use min(knn_k, n_valid) for each query
    # Sort distances (inf will be at the end)
    z_dists_sorted, _ = torch.sort(z_dists, dim=1)  # (N_l, top_K)
    
    # Take top-k, but handle cases where n_valid < knn_k
    actual_k = torch.clamp(n_valid, min=1, max=knn_k)  # (N_l,)
    
    # Compute mean of top-k distances for each query
    mem_scores = torch.zeros(query_z.shape[0], device=query_z.device)
    for k_val in range(1, knn_k + 1):
        # For queries with actual_k >= k_val, accumulate the k_val-th distance
        mask = actual_k >= k_val
        if mask.any():
            mem_scores[mask] += z_dists_sorted[mask, k_val - 1]
    
    mem_scores = mem_scores / actual_k.float().clamp(min=1)
    
    # Fallback: if no valid neighbors at all, assign high score
    no_neighbor = n_valid == 0
    if no_neighbor.any():
        mem_scores[no_neighbor] = z_dists_sorted[:, 0].max() * 2  # 高于最大距离
    
    return mem_scores

# faiss_index.py — ContextSearcher 返回 valid mask
# ==========================================

class ContextSearcher:
    @staticmethod
    def search(
        query_c: torch.Tensor,
        candidate_c: torch.Tensor,
        candidate_z: torch.Tensor,
        top_k: int,
    ) -> Tuple[torch.Tensor, torch.Tensor, torch.Tensor]:  # ← 新增第三个返回值
        """
        Returns:
            neighbor_z:     (N_q, top_k, d_z)
            neighbor_dists: (N_q, top_k)  
            neighbor_valid: (N_q, top_k) bool  ← 新增
        """
        actual_k = min(top_k, candidate_c.shape[0])
        N_q = query_c.shape[0]
        d_z = candidate_z.shape[1]
        device = query_c.device
        
        if actual_k == 0:
            return (
                torch.zeros(N_q, top_k, d_z, device=device),
                torch.full((N_q, top_k), float('inf'), device=device),
                torch.zeros(N_q, top_k, dtype=torch.bool, device=device),  # all invalid
            )
        
        dists = torch.cdist(query_c, candidate_c, p=2.0)
        topk_dists, topk_idx = torch.topk(dists, actual_k, dim=1, largest=False)
        neighbor_z = candidate_z[topk_idx]
        
        # Build valid mask
        valid = torch.ones(N_q, actual_k, dtype=torch.bool, device=device)
        
        # Pad if needed
        if actual_k < top_k:
            pad_n = top_k - actual_k
            neighbor_z = torch.cat([
                neighbor_z,
                torch.zeros(N_q, pad_n, d_z, device=device)
            ], dim=1)
            topk_dists = torch.cat([
                topk_dists,
                torch.full((N_q, pad_n), float('inf'), device=device)
            ], dim=1)
            valid = torch.cat([
                valid,
                torch.zeros(N_q, pad_n, dtype=torch.bool, device=device)
            ], dim=1)
        
        return neighbor_z, topk_dists, valid
```

```python
def train_stage_a(self):
    optimizer = AdamW(
        self.model.parameters(),
        lr=self.config.lr,
        weight_decay=self.config.weight_decay,
    )
    
    # ---- 新增：学习率调度器 ----

        optimizer, 
        T_max=self.config.stage_a_epochs, 
        eta_min=self.config.lr * 0.01,
    )
    
    best_loss = float('inf')
    
    for epoch in range(self.config.stage_a_epochs):
        self.model.train()
        epoch_loss = 0
        
        for batch in self.train_loader:
            x = batch['x'].to(self.device)
            loss_dict = self.model.compute_loss(x)
            loss = loss_dict['total']
            
            optimizer.zero_grad()
            loss.backward()
            
            # ---- 新增：梯度裁剪 ----
            torch.nn.utils.clip_grad_norm_(
                self.model.parameters(), max_norm=1.0
            )
            
            optimizer.step()
            epoch_loss += loss.item()
        
        # ---- 新增：调度器步进 ----
        scheduler.step()
        
        avg_loss = epoch_loss / len(self.train_loader)
        
        # ---- 修改：checkpoint 包含完整状态 ----
        if avg_loss < best_loss:
            best_loss = avg_loss
            self.save_checkpoint({
                'model_state': self.model.state_dict(),
                'optimizer_state': optimizer.state_dict(),
                'scheduler_state': scheduler.state_dict(),
                'epoch': epoch,
                'best_loss': best_loss,
                'normalizer_mean': self.normalizer.scaler.mean_,
                'normalizer_std': self.normalizer.scaler.scale_,
```

```python
def train_stage_c(self):
    # ---- 新增：显式冻结 encoder ----
    self.model.eval()
    for name, param in self.model.named_parameters():
        param.requires_grad = False
    
    # 只解冻 scorer
    self.scorer = self.scorer.to(self.device)
    for param in self.scorer.parameters():
        param.requires_grad = True
    
    optimizer = AdamW(self.scorer.parameters(), lr=1e-3)
    
    for epoch in range(self.config.stage_c_epochs):
        for batch in self.train_loader:
            x = batch['x'].to(self.device)
            B = x.shape[0]
            
            with torch.no_grad():
                # 正常样本的特征
                normal_features = self._extract_scorer_features(x)  # (B*N_total, 4)
            
            # ---- 修改：按 pseudo_anomaly_ratio 生成伪异常 ----
            n_patches_total = normal_features.shape[0]
            n_corrupt = max(1, int(n_patches_total * self.config.pseudo_anomaly_ratio))
            
            # 随机选择要污染的 patch 索引
            corrupt_indices = torch.randperm(n_patches_total)[:n_corrupt]
            
            with torch.no_grad():
                anomaly_features = self._extract_scorer_features_with_corruption(
                    x, corrupt_indices
                )
            
            # 构造标签
            labels = torch.zeros(n_patches_total, device=self.device)
            labels[corrupt_indices] = 1.0
            
            # 用真实特征 + 被污染 patch 的特征拼接
            features = normal_features.clone()
            features[corrupt_indices] = anomaly_features[corrupt_indices]
            
            # Scorer forward
            scores = self.scorer(features)
            loss = F.binary_cross_entropy(scores, labels)
            
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()
```
